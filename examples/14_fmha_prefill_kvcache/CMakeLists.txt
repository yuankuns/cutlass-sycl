set(FMHA_STANDALONE_PAGED_HD256_NUM_SG "16" CACHE STRING "Paged head_dim=256 subgroups per work-group")

set(FMHA_PREFILL_TILED_Q_128 ${FMHA_STANDALONE_PAGED_HD128_TILE_Q})
set(FMHA_PREFILL_TILED_KV_128 ${FMHA_STANDALONE_PAGED_HD128_TILE_KV})

set(FMHA_PREFILL_TEMPLATE
    "${FMHA_PREFILL_SRC_DIR}/sycl/kernels/flash_attention_v2/xe_fmha_fwd_prefill_kernel.cpp.in")
set(FMHA_PREFILL_NOPAGE_TEMPLATE
    "${FMHA_PREFILL_SRC_DIR}/sycl/kernels/flash_attention_v2/xe_fmha_fwd_prefill_nopage_kernel.cpp.in")
set(FMHA_PREFILL_FP8_TEMPLATE
    "${FMHA_PREFILL_SRC_DIR}/sycl/kernels/flash_attention_v2/xe_fmha_fwd_prefill_fp8_kernel.cpp.in")
set(FMHA_PREFILL_GENERATED)
set(FMHA_STANDALONE_HEAD_DIM_DEFS)
set(FMHA_STANDALONE_PAGED_HEAD_DIM_DEFS)
set(FMHA_STANDALONE_NP_HEAD_DIM_DEFS)

foreach(HEAD_DIM ${FMHA_STANDALONE_HEAD_DIMS})
  if(HEAD_DIM IN_LIST FMHA_PREFILL_PAGED_HEAD_DIMS)
    set(EMIT_PAGED TRUE)
    set(TILED_Q ${FMHA_PREFILL_TILED_Q_${HEAD_DIM}})
    set(TILED_KV ${FMHA_PREFILL_TILED_KV_${HEAD_DIM}})
    set(NUM_SG ${FMHA_PREFILL_NUM_SG_${HEAD_DIM}})
    if(NOT TILED_Q OR NOT TILED_KV OR NOT NUM_SG)
      message(FATAL_ERROR "Missing paged tile params for prefill HEAD_DIM=${HEAD_DIM}")
    endif()
    math(EXPR TILED_OUT "((${HEAD_DIM} + 31) / 32) * 32")
    if(DEFINED FMHA_PREFILL_TILED_OUT_${HEAD_DIM})
      set(TILED_OUT ${FMHA_PREFILL_TILED_OUT_${HEAD_DIM}})
    endif()
    if(HEAD_DIM STREQUAL "128" AND FMHA_STANDALONE_PAGED_HD128_LARGE_TILE)
      set(HD128_PAGED_LARGE_TILE 1)
    else()
      set(HD128_PAGED_LARGE_TILE 0)
    endif()
    set(HD128_PAGED_LARGE_TILE_MIN_Q ${FMHA_STANDALONE_PAGED_HD128_LARGE_TILE_MIN_Q})
    set(HD128_PAGED_LARGE_TILE_Q ${FMHA_STANDALONE_PAGED_HD128_LARGE_TILE_Q})
    set(HD128_PAGED_LARGE_TILE_KV ${FMHA_STANDALONE_PAGED_HD128_LARGE_TILE_KV})
    set(HD128_PAGED_LARGE_NUM_SG ${FMHA_STANDALONE_PAGED_HD128_LARGE_NUM_SG})

    set(GENERATED_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/xe_fmha_fwd_prefill_paged_kernel_${HEAD_DIM}.cpp")
    configure_file("${FMHA_PREFILL_TEMPLATE}" "${GENERATED_FILE}" @ONLY)
    list(APPEND FMHA_PREFILL_GENERATED "${GENERATED_FILE}")

    set(GENERATED_FP8_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/xe_fmha_fwd_prefill_fp8_kernel_${HEAD_DIM}.cpp")
    configure_file("${FMHA_PREFILL_FP8_TEMPLATE}" "${GENERATED_FP8_FILE}" @ONLY)
    list(APPEND FMHA_PREFILL_GENERATED "${GENERATED_FP8_FILE}")
    list(APPEND FMHA_STANDALONE_PAGED_HEAD_DIM_DEFS "FMHA_STANDALONE_HAS_PAGED_HD_${HEAD_DIM}=1")
  else()
    set(EMIT_PAGED FALSE)
  endif()

  if(HEAD_DIM IN_LIST FMHA_PREFILL_NP_HEAD_DIMS)
    set(EMIT_NP TRUE)
    set(TILED_Q_NP ${FMHA_PREFILL_TILED_Q_NP_${HEAD_DIM}})
    set(TILED_KV_NP ${FMHA_PREFILL_TILED_KV_NP_${HEAD_DIM}})
    set(NUM_SG_NP ${FMHA_PREFILL_NUM_SG_NP_${HEAD_DIM}})
    if(NOT TILED_Q_NP OR NOT TILED_KV_NP OR NOT NUM_SG_NP)
      message(FATAL_ERROR "Missing non-paged tile params for prefill HEAD_DIM=${HEAD_DIM}")
    endif()
    if(HEAD_DIM STREQUAL "96" AND FMHA_STANDALONE_NP_HD96_SMALL_TILE)
      set(HD96_NP_SMALL_TILE 1)
    else()
      set(HD96_NP_SMALL_TILE 0)
    endif()
    set(HD96_NP_SMALL_TILE_MAX_Q ${FMHA_STANDALONE_NP_HD96_SMALL_TILE_MAX_Q})
    set(HD96_NP_SMALL_TILE_Q ${FMHA_STANDALONE_NP_HD96_SMALL_TILE_Q})
    set(HD96_NP_SMALL_TILE_KV ${FMHA_STANDALONE_NP_HD96_SMALL_TILE_KV})
    set(HD96_NP_SMALL_NUM_SG ${FMHA_STANDALONE_NP_HD96_SMALL_NUM_SG})
    if(HEAD_DIM STREQUAL "96" AND FMHA_STANDALONE_NP_HD96_LARGE_TILE)
      set(HD96_NP_LARGE_TILE 1)
    else()
      set(HD96_NP_LARGE_TILE 0)
    endif()
    set(HD96_NP_LARGE_TILE_MIN_Q ${FMHA_STANDALONE_NP_HD96_LARGE_TILE_MIN_Q})
    set(HD96_NP_LARGE_TILE_Q ${FMHA_STANDALONE_NP_HD96_LARGE_TILE_Q})
    set(HD96_NP_LARGE_TILE_KV ${FMHA_STANDALONE_NP_HD96_LARGE_TILE_KV})
    set(HD96_NP_LARGE_NUM_SG ${FMHA_STANDALONE_NP_HD96_LARGE_NUM_SG})

    set(GENERATED_NP_FILE "${CMAKE_CURRENT_BINARY_DIR}/generated/xe_fmha_fwd_prefill_nopage_kernel_${HEAD_DIM}.cpp")
    configure_file("${FMHA_PREFILL_NOPAGE_TEMPLATE}" "${GENERATED_NP_FILE}" @ONLY)
    list(APPEND FMHA_PREFILL_GENERATED "${GENERATED_NP_FILE}")
    list(APPEND FMHA_STANDALONE_NP_HEAD_DIM_DEFS "FMHA_STANDALONE_HAS_NP_HD_${HEAD_DIM}=1")
  else()
    set(EMIT_NP FALSE)
  endif()

  if(NOT EMIT_PAGED AND NOT EMIT_NP)
    message(FATAL_ERROR "Unsupported standalone FMHA prefill HEAD_DIM=${HEAD_DIM}")
  endif()
  list(APPEND FMHA_STANDALONE_HEAD_DIM_DEFS "FMHA_STANDALONE_HAS_HD_${HEAD_DIM}=1")
endforeach()

list(REMOVE_DUPLICATES FMHA_STANDALONE_HEAD_DIM_DEFS)

add_library(fmha_prefill_kvcache_kernels STATIC
  ${FMHA_PREFILL_GENERATED}
  standalone_runtime.cpp
)

target_compile_features(fmha_prefill_kvcache_kernels PUBLIC cxx_std_20)
target_compile_definitions(fmha_prefill_kvcache_kernels PUBLIC SGL_KERNEL_STANDALONE_NO_TORCH)
target_compile_options(fmha_prefill_kvcache_kernels PRIVATE
  $<$<COMPILE_LANGUAGE:CXX>:${FMHA_STANDALONE_OPT_LEVEL}>
)
target_include_directories(fmha_prefill_kvcache_kernels PUBLIC
  "${FMHA_PREFILL_EXAMPLE_ROOT}"
  "${FMHA_PREFILL_SRC_DIR}"
  "${CUTLASS_EXAMPLES_COMMON_SOURCE_DIR}"
  "${CUTLASS_EXAMPLES_UTILS_DIR}"
  "${CUTLASS_APPLICATIONS_DIR}"
)
target_link_libraries(fmha_prefill_kvcache_kernels PUBLIC
  CUTLASS
  cutlass_tools_util_includes
)

add_sycl_to_target(TARGET fmha_prefill_kvcache_kernels)

  ${FMHA_STANDALONE_HEAD_DIM_DEFS}
  ${FMHA_STANDALONE_PAGED_HEAD_DIM_DEFS}
  ${FMHA_STANDALONE_NP_HEAD_DIM_DEFS}
)
target_include_directories(fmha_prefill_kvcache PRIVATE
  "${FMHA_PREFILL_EXAMPLE_ROOT}"
  "${FMHA_PREFILL_SRC_DIR}"
  "${CUTLASS_EXAMPLES_COMMON_SOURCE_DIR}"
  "${CUTLASS_EXAMPLES_UTILS_DIR}"
  "${CUTLASS_APPLICATIONS_DIR}"
)
target_link_libraries(fmha_prefill_kvcache PRIVATE fmha_prefill_kvcache_kernels)

add_sycl_to_target(TARGET fmha_prefill_kvcache)

  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
